DROP TABLE IF EXISTS BRANCH;
DROP TABLE IF EXISTS STAFF;
DROP TABLE IF EXISTS PROPERTY;
DROP TABLE IF EXISTS RENTER;
DROP TABLE IF EXISTS OWNER;
DROP TABLE IF EXISTS VIEWING;


CREATE TABLE BRANCH (
    BranchNo VARCHAR(3) PRIMARY KEY,
    Street VARCHAR(100),
    Area VARCHAR(50),
    City VARCHAR(50),
    TelNo VARCHAR(20)
);

CREATE TABLE STAFF (
	StaffNo INT PRIMARY KEY,
	FName VARCHAR(20),
	LName VARCHAR(20),
	Address VARCHAR(100),
	TelNo VARCHAR(20),
	Job VARCHAR(40),
	Gender CHAR(20),
	Salary DECIMAL(20,2),
	BranchNo VARCHAR(3)
);

CREATE TABLE PROPERTY (
	PropertyNo INT PRIMARY KEY,
	Street VARCHAR(100),
	Area VARCHAR(50),
	City VARCHAR(50),
	Type VARCHAR(20),
	Rooms INT,
	Rent DECIMAL(10,2),
	OwnerNo INT,
	StaffNo INT,
	BranchNo VARCHAR(3)
);

CREATE TABLE RENTER (
	RenterNo INT PRIMARY KEY,
	FName VARCHAR(20),
	LName VARCHAR(20),
	Address VARCHAR(100),
	TelNo VARCHAR(20),
	TypePreference VARCHAR(20),
	MaxRent DECIMAL(7,2)
);

CREATE TABLE OWNER (
	OwnerNo INT PRIMARY KEY,
	FName VARCHAR(20),
	LName VARCHAR(20),
	Address VARCHAR(100),
	TelNo VARCHAR(20)
);

CREATE TABLE VIEWING (
	RenterNo INT,
	PropertyNo INT,
	Date DATE,
	Comment VARCHAR(100)
);

SELECT R.FName, R.LName, R.Address FROM RENTER as R JOIN VIEWING as V ON 
R.RenterNo = V.RenterNo;

-- Question 2
SELECT S.FName, S.LName, S.BranchNo FROM STAFF as S JOIN PROPERTY as P ON 
S.StaffNo = P.StaffNo WHERE P.City LIKE "Cork%";

-- Question 3
SELECT R.FName, R.LName FROM RENTER as R JOIN VIEWING as V ON 
R.RenterNo = V.RenterNo JOIN PROPERTY as P ON P.PropertyNo = 
V.PropertyNo WHERE P.Rooms >= 3;

-- Question 4 
SELECT DISTINCT R1.FName, R1.LName, R1.TelNo, R2.FName, R2.LName, R2.TelNo FROM RENTER as R1 
JOIN RENTER as R2 ON R1.Address = R2.Address AND R1.RenterNo <> R2.RenterNo GROUP BY R1.Address
HAVING COUNT(*) > 1;

-- Question 5
SELECT S.FName, S.LName FROM STAFF as S JOIN RENTER as R ON 
R.FName = S.FName AND R.LName = S.LName JOIN OWNER as O ON
R.FName = O.FName AND R.LName = O.LName WHERE R.FName = O.FName AND 
O.FName = S.FName AND R.LName = O.LName AND O.LName = S.LName;
